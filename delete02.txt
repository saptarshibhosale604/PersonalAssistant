#!/bin/bash

# Update system
sudo apt update
sudo apt upgrade -y

# Install prerequisites
sudo apt install -y zsh git curl fonts-powerline

# Install Oh My Zsh
sh -c "$(curl -fsSL https://raw.githubusercontent.com/ohmyzsh/ohmyzsh/master/tools/install.sh)" --unattended

# Change default shell to zsh (requires logout/login or exec)
chsh -s $(which zsh)

# Install zsh-autosuggestions (auto-completion/suggestions)
if [ ! -d "${ZSH_CUSTOM:-~/.oh-my-zsh/custom}/plugins/zsh-autosuggestions" ]; then
  git clone https://github.com/zsh-users/zsh-autosuggestions ${ZSH_CUSTOM:-~/.oh-my-zsh/custom}/plugins/zsh-autosuggestions
fi

# Install zsh-syntax-highlighting
if [ ! -d "${ZSH_CUSTOM:-~/.oh-my-zsh/custom}/plugins/zsh-syntax-highlighting" ]; then
  git clone https://github.com/zsh-users/zsh-syntax-highlighting.git ${ZSH_CUSTOM:-~/.oh-my-zsh/custom}/plugins/zsh-syntax-highlighting
fi

# Enable plugins in .zshrc
sed -i 's/^plugins=(git)/plugins=(git zsh-autosuggestions zsh-syntax-highlighting)/' ~/.zshrc

# Reload zsh config
source ~/.zshrc

echo "Installation complete! Log out and log back in for full effect, or run 'exec zsh'."
echo "Auto-suggestions and syntax highlighting are now enabled (likely what you meant by 'starting auto compiler')."
